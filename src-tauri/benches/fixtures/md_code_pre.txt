### 9. SQL

```sql
SELECT 
    u.username,
    COUNT(m.id) AS message_count,
    MAX(m.created_at) AS last_message
FROM users u
LEFT JOIN messages m ON u.id = m.user_id
WHERE u.active = true
GROUP BY u.username
HAVING COUNT(m.id) > 0
ORDER BY message_count DESC
LIMIT 10;
```

### 12. TypeScript 带泛型

```typescript
interface StreamBuffer<T> {
    data: T[];
    threshold: number;
    flush(): T[];
}

function createBuffer<T>(threshold: number): StreamBuffer<T> {
    return {
        data: [],
        threshold,
        flush() {
            const chunk = this.data;
            this.data = [];
            return chunk;
        }
    };
}
```
